CREATE OR REPLACE PROCEDURE `real_estates.log_property_change`(
  announcement_id STRING,
  field_changed STRING,
  old_value STRING,
  new_value STRING
)
BEGIN
  INSERT INTO `real_estates.properties_announcements_changes` (
    change_id,
    announcement_id,
    field_changed,
    old_value,
    new_value,
    change_timestamp,
    ack
  )
  VALUES (
    GENERATE_UUID(),
    announcement_id,
    field_changed,
    old_value,
    new_value,
    CURRENT_TIMESTAMP(),
    False
  );
END;

CREATE OR REPLACE VIEW `real_estates.price_history` AS
SELECT
  announcement_id,
  price AS new_value,
  created_at AS change_timestamp
FROM
  `real_estates.properties_announcements`
UNION ALL
SELECT
  announcement_id,
  CAST(new_value AS FLOAT64) AS new_value,
  change_timestamp
FROM
  `real_estates.properties_announcements_changes`
WHERE
  field_changed = 'price'
ORDER BY
  announcement_id, change_timestamp;
